    52→function makeLogoSvg(name: string, primary: string, variant: 'cross' | 'arch'): string {
    53→  const initials = name
    54→    .split(' ')
    55→    .filter((w) => w.length > 2)
    56→    .slice(0, 2)
    57→    .map((w) => w[0])
    58→    .join('')
    59→  const mark =
    60→    variant === 'cross'
    61→      ? `<rect x="46" y="18" width="8" height="64" rx="2" fill="${primary}"/><rect x="28" y="38" width="44" height="8" rx="2" fill="${primary}"/>`
    62→      : `<path d="M50 14 C30 14 18 28 18 46 L18 86 L82 86 L82 46 C82 28 70 14 50 14 Z" fill="none" stroke="${primary}" stroke-width="7"/><rect x="44" y="8" width="12" height="22" rx="3" fill="${primary}"/>`
    63→  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100" role="img" aria-label="${name} logo">
    64→  <rect width="100" height="100" rx="20" fill="none"/>
    65→  ${mark}
    66→  <text x="50" y="72" text-anchor="middle" font-family="Georgia, serif" font-size="13" fill="${primary}">${initials}</text>
    67→</svg>`
    68→}

    70→async function putAsset(opts: {
    71→  churchId: string
    72→  fileName: string
    73→  buffer: Buffer
    74→  mimeType: string
    75→  kind: 'IMAGE' | 'DOCUMENT'
    76→  altText: string
    77→  uploadedById: string
    78→  folder?: string
    79→}) {
    80→  const storedName = `${randomUUID().replace(/-/g, '').slice(0, 16)}-${opts.fileName
    81→    .toLowerCase()
    82→    .replace(/[^a-z0-9._-]/g, '-')
    83→    .slice(-40)}`
    84→  const key = `${opts.churchId}/${storedName}`
    85→  const full = path.join(STORAGE_ROOT, key)
    86→  await mkdir(path.dirname(full), { recursive: true })
    87→  await writeFile(full, opts.buffer)
    88→  return db.mediaAsset.create({
    89→    data: {
    90→      churchId: opts.churchId,
    91→      fileName: opts.fileName,
    92→      storedName,
    93→      mimeType: opts.mimeType,
    94→      kind: opts.kind,
    95→      sizeBytes: opts.buffer.byteLength,
    96→      altText: opts.altText,
    97→      folder: opts.folder ?? 'Demo content',
    98→      visibility: 'PUBLIC',
    99→      uploadedById: opts.uploadedById,
   100→    },
   101→  })
   102→}

   104→async function download(url: string): Promise<{ buffer: Buffer; mimeType: string; fileName: string }> {
   105→  const res = await fetch(url)
   106→  if (!res.ok) throw new Error(`Failed to fetch ${url}: ${res.status}`)
   107→  const buffer = Buffer.from(await res.arrayBuffer())
   108→  const mimeType = res.headers.get('content-type') ?? 'image/jpeg'
   109→  const ext = mimeType.includes('webp') ? 'webp' : mimeType.includes('png') ? 'png' : 'jpg'
   110→  const fileName = url.split('/').pop()?.split('.')[0] + '.' + ext
   111→  return { buffer, mimeType: mimeType.split(';')[0], fileName }
   112→}

   116→/** The next upcoming Remembrance Sunday (2nd Sunday of November), 10:00 UTC. */
   117→function nextRemembranceSunday(): Date {
   118→  const forYear = (year: number) => {
   119→    const d = new Date(Date.UTC(year, 10, 1))
   120→    // First Sunday of November: advance (7 - weekday) % 7 days — 0 when
   121→    // 1 November is already a Sunday (the old `|| 7` skipped a week and
   122→    // produced the 4th Sunday instead of the 2nd).
   123→    d.setUTCDate(d.getUTCDate() + ((7 - d.getUTCDay()) % 7))
   124→    const secondSunday = new Date(d.getTime() + 7 * 24 * 60 * 60 * 1000)
   125→    secondSunday.setUTCHours(10, 0, 0, 0) // service starts 10:00 UTC
   126→    return secondSunday
   127→  }
   128→  const now = new Date()
   129→  const thisYear = forYear(now.getFullYear())
   130→  // Service runs 10:00–12:00 UTC; once it has ended, roll to next year's.
   131→  if (new Date(thisYear.getTime() + 2 * 60 * 60 * 1000) <= now) return forYear(now.getFullYear() + 1)
   132→  return thisYear
   133→}

   135→function daysFromNow(days: number, hour = 10, minute = 0): Date {
   136→  const d = new Date()
   137→  d.setDate(d.getDate() + days)
   138→  d.setHours(hour, minute, 0, 0)
   139→  return d
   140→}
